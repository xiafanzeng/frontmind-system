#!/usr/bin/env python3
"""Adversarial tests for the v2.1 Dashboard single-question adapter."""

from __future__ import annotations

import contextlib
import importlib.util
import io
import json
import sys
import tempfile
import unittest
from pathlib import Path
from types import SimpleNamespace
from unittest.mock import patch

from openpyxl import Workbook


sys.dont_write_bytecode = True
SKILL_ROOT = Path(__file__).resolve().parents[1]


def load_script(name: str):
    path = SKILL_ROOT / "scripts" / f"{name}.py"
    spec = importlib.util.spec_from_file_location(f"e2_test_{name}", path)
    if spec is None or spec.loader is None:
        raise RuntimeError(f"cannot load {path}")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


ADAPTER = load_script("dashboard_source_adapter")
QUESTION = "哪家 GEO 服务商值得选？"
NORMALIZED_QUESTION = ADAPTER.normalized_question(QUESTION)
CONTENT_HEADERS = [
    "引用排行", "内容状态", "内容标题", "内容URL", "媒体名称", "媒体url",
    "引用日期", "引用次数", "引用占比",
]
DETAIL_HEADERS = ["引用日期", "模型", "监控问题", "内容标题", "内容链接", "媒体名称", "媒体url"]


class DashboardSourceAdapterTests(unittest.TestCase):
    def setUp(self) -> None:
        self.temporary = tempfile.TemporaryDirectory()
        self.root = Path(self.temporary.name)

    def tearDown(self) -> None:
        self.temporary.cleanup()

    def write_workbook(
        self,
        rows: list[dict[str, object]],
        *,
        question_overrides: dict[tuple[int, int], str] | None = None,
    ) -> Path:
        workbook = Workbook()
        content = workbook.active
        content.title = "内容统计"
        content.append(CONTENT_HEADERS)
        details = workbook.create_sheet("详细表格")
        details.append(DETAIL_HEADERS)
        guide = workbook.create_sheet("字段说明")
        guide.append(["字段名称", "字段释义"])
        for field in ("引用排行", "引用占比", "内容标题", "内容URL", "引用次数", "内容状态"):
            if field == "引用排行":
                definition = "按被引次数排序，排名靠前的是AI主要知识源。"
            elif field == "引用占比":
                definition = "该信源引用次数占总引用次数的百分比，衡量其影响力大小。"
            else:
                definition = f"Dashboard {field}"
            guide.append([field, definition])

        overrides = question_overrides or {}
        for row_index, item in enumerate(rows):
            content.append([
                item["rank"], item.get("status", "已收录"), item.get("title", f"页面 {item['rank']}"),
                item["url"], item.get("media", "测试媒体"), item.get("media_url", "media.example/path"),
                item.get("date", "2026-08-01 - 2026-08-02"), item["count"], item["share"],
            ])
            for citation_index in range(int(item.get("detail_count", item["count"]))):
                detail_question = overrides.get((row_index, citation_index), QUESTION)
                details.append([
                    "2026-08-01", "test-model", detail_question, item.get("title", f"页面 {item['rank']}"),
                    item["url"], item.get("media", "测试媒体"), item.get("media_url", "media.example/path"),
                ])
        path = self.root / "01_monitoring" / "dashboard_source_workbook.xlsx"
        path.parent.mkdir(parents=True, exist_ok=True)
        workbook.save(path)
        workbook.close()
        return path

    def run_adapter(self, workbook: Path) -> dict[str, object]:
        content_job = {
            "schema_version": "2.1.0",
            "job_id": "job_e2_dashboard_test",
            "pack_id": "rp_e2_dashboard_test",
            "created_at": "2026-08-12T00:00:00+00:00",
            "entry_category": "industry_ranking",
            "task": {
                "question_text": NORMALIZED_QUESTION,
                "question_origin": "external_confirmed",
                "foundation_subject": None,
                "manual_foundation_selection": False,
                "product_scenario_subintent": None,
            },
            "scope_analysis_path": "00_input/scope_analysis.json",
            "monitoring_input_path": "01_monitoring/monitoring_input.json",
            "source_workbook_path": "01_monitoring/dashboard_source_workbook.xlsx",
            "e2_pattern_analysis_path": "02_pattern/E2_pattern_analysis.json",
            "optimizer_target": {"dataset": "fixture", "engine_llm": "fixture"},
            "deliverables": {"docx_body": True, "html_body_fragment": True, "title_map": True},
            "editorial_profile": "third_party_objective_with_evidence_based_brand_promotion",
        }
        content_job_path = self.root / "content_job.json"
        content_job_path.write_text(json.dumps(content_job, ensure_ascii=False), encoding="utf-8")
        output = self.root / "source-pool.json"
        argv = [
            "dashboard_source_adapter.py", "--content-job", str(content_job_path),
            "--workbook", str(workbook), "--package-root", str(self.root),
            "--output", str(output),
        ]
        with patch.object(sys, "argv", argv), contextlib.redirect_stdout(io.StringIO()):
            ADAPTER.main()
        return json.loads(output.read_text(encoding="utf-8"))

    def test_accepts_exact_normalized_single_question_and_derives_media_domain(self) -> None:
        workbook = self.write_workbook(
            [{
                "rank": 1, "url": "https://article.example/a", "count": 1, "share": "100.00%",
                "media_url": "HTTPS://News.Example.COM/channel?id=1",
            }],
            question_overrides={(0, 0): "  哪家   GEO 服务商值得选？  "},
        )

        payload = self.run_adapter(workbook)

        self.assertEqual(payload["question"], NORMALIZED_QUESTION)
        self.assertEqual(payload["cited_content_pool"][0]["media_domain"], "news.example.com")
        self.assertEqual(payload["cited_content_pool"][0]["detailed_row_refs"], [2])
        self.assertEqual(payload["source_workbook_receipt"]["dashboard_total_citation_count"], 1)
        self.assertEqual(payload["source_workbook_receipt"]["top20_citation_count"], 1)
        self.assertEqual(
            payload["source_workbook_receipt"]["citation_rank_definition"],
            "citation_count_descending",
        )
        self.assertTrue(payload["source_workbook_receipt"]["content_status_ignored"])
        self.assertNotIn("内容状态", json.dumps(payload, ensure_ascii=False))

    def test_effective_range_falls_back_when_ooxml_dimension_metadata_is_missing(self) -> None:
        rows = [
            (SimpleNamespace(value="A"), SimpleNamespace(value="B")),
            (SimpleNamespace(value="x"), SimpleNamespace(value=None)),
        ]
        sheet = SimpleNamespace(max_row=None, max_column=None, iter_rows=lambda: iter(rows))
        self.assertEqual(ADAPTER.effective_range(sheet), "A1:B2")

    def test_rejects_a_workbook_containing_any_second_question(self) -> None:
        workbook = self.write_workbook(
            [{"rank": 1, "url": "https://article.example/a", "count": 2, "share": 1}],
            question_overrides={(0, 1): "哪家 AEO 服务商值得选？"},
        )

        with self.assertRaisesRegex(ValueError, "prefiltered to exactly the Content Job question"):
            self.run_adapter(workbook)

    def test_rejects_a_gap_in_raw_top20_ranks(self) -> None:
        workbook = self.write_workbook([
            {"rank": 1, "url": "https://article.example/a", "count": 1, "share": 0.5},
            {"rank": 3, "url": "https://article.example/c", "count": 1, "share": 0.5},
        ])

        with self.assertRaisesRegex(ValueError, "contiguous 1..N without gaps"):
            self.run_adapter(workbook)

    def test_rejects_duplicate_raw_top20_rank_even_when_urls_differ(self) -> None:
        workbook = self.write_workbook([
            {"rank": 1, "url": "https://article.example/a", "count": 1, "share": 0.5},
            {"rank": 1, "url": "https://article.example/b", "count": 1, "share": 0.5},
        ])

        with self.assertRaisesRegex(ValueError, "duplicate 引用排行 1"):
            self.run_adapter(workbook)

    def test_rank21_never_backfills_a_short_raw_pool(self) -> None:
        workbook = self.write_workbook([
            {"rank": 1, "url": "https://article.example/a", "count": 1, "share": 0.5},
            {"rank": 21, "url": "https://article.example/should-not-backfill", "count": 1, "share": 0.5},
        ])

        payload = self.run_adapter(workbook)

        self.assertEqual(payload["pool_status"], "partial_pool")
        self.assertEqual([item["raw_rank"] for item in payload["cited_content_pool"]], [1])
        self.assertEqual(payload["source_workbook_receipt"]["dashboard_total_citation_count"], 2)
        self.assertEqual(payload["source_workbook_receipt"]["top20_citation_count"], 1)
        self.assertNotIn("should-not-backfill", json.dumps(payload, ensure_ascii=False))

    def test_duplicate_rank_after_top20_is_still_rejected_before_filtering(self) -> None:
        workbook = self.write_workbook([
            {"rank": 1, "url": "https://article.example/a", "count": 1, "share": 1 / 3},
            {"rank": 21, "url": "https://article.example/u", "count": 1, "share": 1 / 3},
            {"rank": 21, "url": "https://article.example/v", "count": 1, "share": 1 / 3},
        ])

        with self.assertRaisesRegex(ValueError, "duplicate 引用排行 21"):
            self.run_adapter(workbook)

    def test_rejects_citation_count_mismatch_against_exact_question_details(self) -> None:
        workbook = self.write_workbook([
            {
                "rank": 1, "url": "https://article.example/a", "count": 2,
                "detail_count": 1, "share": 1,
            },
        ])

        with self.assertRaisesRegex(ValueError, "count mismatch"):
            self.run_adapter(workbook)

    def test_rejects_top20_url_without_matching_detail_record(self) -> None:
        workbook_path = self.write_workbook([
            {"rank": 1, "url": "https://article.example/a", "count": 1, "share": 1},
        ])
        from openpyxl import load_workbook

        workbook = load_workbook(workbook_path)
        workbook["详细表格"]["E2"] = "https://article.example/different"
        workbook.save(workbook_path)
        workbook.close()

        with self.assertRaisesRegex(ValueError, "has no exact-question 详细表格 record"):
            self.run_adapter(workbook_path)

    def test_rejects_citation_share_mismatch_against_exact_question_total(self) -> None:
        workbook = self.write_workbook([
            {"rank": 1, "url": "https://article.example/a", "count": 1, "share": 0.6},
            {"rank": 2, "url": "https://article.example/b", "count": 1, "share": 0.4},
        ])

        with self.assertRaisesRegex(ValueError, "citation_share does not reconcile"):
            self.run_adapter(workbook)

    def test_rejects_incorrect_field_guide_semantics(self) -> None:
        workbook_path = self.write_workbook([
            {"rank": 1, "url": "https://article.example/a", "count": 1, "share": 1},
        ])
        from openpyxl import load_workbook

        workbook = load_workbook(workbook_path)
        guide = workbook["字段说明"]
        for row in guide.iter_rows(min_row=2):
            if row[0].value == "引用占比":
                row[1].value = "这是一个百分比字段"
        workbook.save(workbook_path)
        workbook.close()

        with self.assertRaisesRegex(ValueError, "字段说明.引用占比"):
            self.run_adapter(workbook_path)

    def test_content_status_changes_do_not_change_canonical_source_pool_semantics(self) -> None:
        workbook_path = self.write_workbook([
            {
                "rank": 1, "url": "https://article.example/a", "count": 1,
                "share": 1, "status": "已收录",
            },
        ])
        first = self.run_adapter(workbook_path)
        (self.root / "source-pool.json").unlink()

        from openpyxl import load_workbook

        workbook = load_workbook(workbook_path)
        workbook["内容统计"]["B2"] = "任意完全不同的目录状态"
        workbook.save(workbook_path)
        workbook.close()
        second = self.run_adapter(workbook_path)

        for payload in (first, second):
            payload.pop("created_at")
            payload["source_workbook_receipt"].pop("sha256")
        self.assertEqual(first, second)
