def load_script(name: str, path: Path):
    spec = importlib.util.spec_from_file_location(f"e3_v21_test_{name}", path)
    if spec is None or spec.loader is None:
        raise RuntimeError(f"cannot load {path}")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def write_json(path: Path, value: dict[str, object]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2), encoding="utf-8")


def signed_registry() -> dict[str, object]:
    patterns: list[dict[str, object]] = []
    for number in range(1, 17):
        pattern_id = f"P{number:02d}"
        component_id = f"{pattern_id}_component_core"
        contract: dict[str, object] = {
            "eligible_content_forms": ["comparison_review"],
            "positive_intent_signals": [],
            "typical_structure_signals": [],
            "exclusion_signals": [],
            "confusable_patterns": [],
        }
        if pattern_id == "P01":
            contract.update({
                "positive_intent_signals": [
                    {"signal_id": "P01_intent_rank", "cues": ["服务商排行"]},
                ],
                "typical_structure_signals": [
                    {"component_id": component_id, "cues": ["评价标准"]},
                ],
            })
        patterns.append({
            "id": pattern_id,
            "trigger_signals": ["值得选"] if pattern_id == "P01" else [],
            "structure_components": [{"component_id": component_id}],
            "classification_contract": contract,
        })
    return {
        "schema_version": "2.1.0",
        "routing": {
            "industry_ranking": {
                "primary_patterns": ["P01"],
                "secondary_patterns": [],
            },
            "foundation_start": {
                "primary_patterns": ["P14"],
                "secondary_patterns": [],
            },
        },
        "patterns": patterns,
    }


def signed_research() -> dict[str, object]:

                "structure_components": [

    def _run_intake(
        self, report_name: str, *, job: dict[str, object] | None = None,
        forbid_e2_validation: bool = False,
    ) -> tuple[int, dict[str, object]]:
        selected_job = copy.deepcopy(job or self._ordinary_job())
        write_json(self.job_path, selected_job)
        report_path = self.root / "03_intake" / report_name
        argv = [
            "intake_validator.py",
            "--content-job", str(self.job_path),
            "--reference-pack", str(self.reference_pack_path),
            "--scope-analysis", str(self.scope_path),
            "--staging-receipt", str(self.staging_receipt_path),
            "--source-zip", str(self.source_zip),
            "--package-root", str(self.root),
            "--report", str(report_path),
        ]
        if selected_job["entry_category"] != "foundation_start":
            argv.extend([
                "--monitoring-input", str(self.monitoring_path),
                "--source-workbook", str(self.workbook_path),
                "--e2-pattern-analysis", str(self.e2_path),
            ])
        registries = {
            name: {"registry_type": name}
            for name in ("knowledge_registry", "source_registry", "claim_registry", "image_registry")
        }
        with ExitStack() as stack:
            stack.enter_context(patch.object(sys, "argv", argv))
            stack.enter_context(patch.object(INTAKE, "validate_schema", return_value=None))
            stack.enter_context(patch.object(INTAKE, "validate_root", return_value=None))
            stack.enter_context(patch.object(
                INTAKE, "run_root_pack_validator", return_value={"status": "pass", "errors": []},
            ))
            stack.enter_context(patch.object(INTAKE, "validate_pack", return_value=(registries, {})))
            if forbid_e2_validation:
                stack.enter_context(patch.object(
                    INTAKE, "validate_e2_relations",
                    side_effect=AssertionError("foundation_start must not call E2 validation"),
                ))
            with contextlib.redirect_stdout(io.StringIO()):
                result = INTAKE.main()
        return result, json.loads(report_path.read_text(encoding="utf-8"))

    def _human_reference_paths(self) -> list[Path]:
        receipt = json.loads(self.human_receipt_path.read_text(encoding="utf-8"))
        return [
            self.root / receipt[key]["path"]
            for key in (
                "automatic_page_analysis_ref", "review_input_ref", "reviewed_page_analysis_ref",
            )
        ]

    def test_ordinary_report_freezes_every_e2_and_monitoring_artifact(self) -> None:
        result, report = self._run_intake("ordinary-pass.json")
        self.assertEqual(result, 0, report["errors"])
        self.assertEqual(report["status"], "passed")

        analysis = json.loads(self.e2_path.read_text(encoding="utf-8"))
        observation = analysis["content_observations"][0]
        snapshot_path = self.root / observation["retrieval_receipt"]["snapshot_relative_path"]
        extract_path = self.root / observation["retrieval_receipt"]["extract_relative_path"]
        expected_frozen_paths = {
            self.workbook_path,
            snapshot_path,
            extract_path,
            self.human_receipt_path,
            *self._human_reference_paths(),
        }
        frozen = report["e2_frozen_artifact_sha256"]
        self.assertEqual(
            set(frozen),
            {path.relative_to(self.root).as_posix() for path in expected_frozen_paths},
        )
        for path in expected_frozen_paths:
            relative = path.relative_to(self.root).as_posix()
            self.assertEqual(frozen[relative], INTAKE.sha256(path))

        self.assertEqual(report["source_workbook_sha256"], INTAKE.sha256(self.workbook_path))
        self.assertEqual(report["monitoring_input_file_sha256"], INTAKE.sha256(self.monitoring_path))
        self.assertEqual(
            report["monitoring_input_sha256"],
            INTAKE.canonical_sha256(json.loads(self.monitoring_path.read_text(encoding="utf-8"))),
        )
        self.assertEqual(report["e2_pattern_analysis_sha256"], INTAKE.sha256(self.e2_path))
        self.assertEqual(
            report["e2_pattern_analysis_canonical_sha256"],
            INTAKE.canonical_sha256(json.loads(self.e2_path.read_text(encoding="utf-8"))),
        )
        self.assertEqual(report["content_pattern_registry_sha256"], INTAKE.sha256(self.registry_path))
        self.assertEqual(report["pattern_research_index_sha256"], INTAKE.sha256(self.research_path))

    def test_tampering_any_bound_input_or_receipt_link_fails_intake(self) -> None:
        analysis = json.loads(self.e2_path.read_text(encoding="utf-8"))
        observation = analysis["content_observations"][0]
        snapshot_path = self.root / observation["retrieval_receipt"]["snapshot_relative_path"]
        extract_path = self.root / observation["retrieval_receipt"]["extract_relative_path"]
        automatic_path, review_input_path, reviewed_path = self._human_reference_paths()

        append_only_targets = {
            "dashboard_workbook": self.workbook_path,
            "registry": self.registry_path,
            "research_index": self.research_path,
            "snapshot": snapshot_path,
            "extract": extract_path,
            "human_receipt": self.human_receipt_path,
            "review_automatic_ref": automatic_path,
            "review_input_ref": review_input_path,
            "reviewed_page_ref": reviewed_path,
            "monitoring_answers_table": self.answers_table,
        }
        for label, path in append_only_targets.items():
            with self.subTest(artifact=label):
                original = path.read_bytes()
                try:
                    path.write_bytes(original + b"\n")
                    result, report = self._run_intake(f"tampered-{label}.json")
                    self.assertEqual(result, 1, report["errors"])
                    self.assertEqual(report["status"], "failed")
                    self.assertTrue(report["errors"], label)
                finally:
                    path.write_bytes(original)

        with self.subTest(artifact="monitoring_input"):
            original = self.monitoring_path.read_bytes()
            try:
                monitoring = json.loads(original)
                monitoring["question"] = "被篡改的另一个问题"
                write_json(self.monitoring_path, monitoring)
                result, report = self._run_intake("tampered-monitoring-input.json")
                self.assertEqual(result, 1, report["errors"])
                self.assertTrue(any("monitoring_input.question" in item for item in report["errors"]))
            finally:
                self.monitoring_path.write_bytes(original)

        with self.subTest(artifact="e2_pattern_analysis"):
            original = self.e2_path.read_bytes()
            try:
                e2 = json.loads(original)
                e2["recommendation"]["advisory_only"] = False
                write_json(self.e2_path, e2)
                result, report = self._run_intake("tampered-e2-analysis.json")
                self.assertEqual(result, 1, report["errors"])
                self.assertTrue(any("advisory_only" in item for item in report["errors"]))
            finally:
                self.e2_path.write_bytes(original)

    def test_foundation_start_skips_monitoring_workbook_and_e2(self) -> None:
        result, report = self._run_intake(
            "foundation-pass.json", job=self._foundation_job(), forbid_e2_validation=True,
        )
        self.assertEqual(result, 0, report["errors"])
        self.assertEqual(report["status"], "passed")
        self.assertEqual(report["checks"]["monitoring_input"], "not_applicable")
        self.assertEqual(report["checks"]["e2_pattern_analysis"], "not_applicable")
        for key in (
            "source_workbook_sha256", "monitoring_input_sha256", "monitoring_input_file_sha256",
            "e2_pattern_analysis_sha256", "e2_pattern_analysis_canonical_sha256",
        ):
            self.assertIsNone(report[key], key)
        self.assertEqual(report["e2_frozen_artifact_sha256"], {})
        self.assertEqual(report["e2_retrieval_artifact_sha256"], {})
